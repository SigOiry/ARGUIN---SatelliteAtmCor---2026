lonlat_to_tile <- function(lon, lat, zoom) {
  n <- 2^zoom
  c(x = floor((lon + 180) / 360 * n),
    y = floor((1 - log(tan(lat*pi/180) + 1/cos(lat*pi/180))/pi) / 2 * n))
}

tile_bounds_3857 <- function(x, y, zoom) {
  world <- 20037508.342789244
  size <- 2 * world / 2^zoom
  c(xmin = -world + x*size, xmax = -world + (x+1)*size,
    ymin = world - (y+1)*size, ymax = world - y*size)
}

get_tile <- function(x, y, zoom, provider = "esri") {
  path <- file.path(tile_dir, sprintf("%s_z%02d_x%d_y%d.png", provider, zoom, x, y))
  if (!file.exists(path) || file.info(path)$size == 0) {
    url <- if (provider == "osm") {
      sprintf("https://tile.openstreetmap.org/%d/%d/%d.png", zoom, x, y)
    } else {
      sprintf("https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/%d/%d/%d", zoom, y, x)
    }
    download.file(url, path, mode = "wb", quiet = TRUE)
  }
  signature <- readBin(path, "raw", n = 2)
  image <- if (identical(signature, as.raw(c(0xff, 0xd8)))) jpeg::readJPEG(path) else png::readPNG(path)
  list(image = image, bounds = tile_bounds_3857(x, y, zoom))
}

tile_layer <- function(plot, x, y, zoom, provider = "esri") {
  tile <- get_tile(x, y, zoom, provider)
  b <- tile$bounds
  plot + annotation_raster(tile$image, b[["xmin"]], b[["xmax"]], b[["ymin"]], b[["ymax"]])
}

tile_plot_for_bbox <- function(bbox, zoom, provider = "esri") {
  ll <- st_transform(st_as_sfc(bbox), 4326) |> st_bbox()
  nw <- lonlat_to_tile(ll[["xmin"]], ll[["ymax"]], zoom)
  se <- lonlat_to_tile(ll[["xmax"]], ll[["ymin"]], zoom)
  p <- ggplot()
  for (x in seq(nw[["x"]], se[["x"]])) for (y in seq(nw[["y"]], se[["y"]])) {
    p <- tile_layer(p, x, y, zoom, provider)
  }
  p
}

make_zoom <- function(ids, panel_letter, cluster_label) {
  s <- stations[stations$station_id %in% ids,]
  s_xy <- cbind(st_drop_geometry(s),as.data.frame(st_coordinates(s)))
  bb <- st_bbox(s)
  half <- 900
  centre <- c(mean(bb[c("xmin","xmax")]), mean(bb[c("ymin","ymax")]))
  bb <- st_bbox(c(xmin=centre[[1]]-half, ymin=centre[[2]]-half,
                  xmax=centre[[1]]+half, ymax=centre[[2]]+half), crs=st_crs(3857))
  tile_plot_for_bbox(bb, 16) +
    geom_sf(data=intertidal,fill=scales::alpha("#2DC7C9",.18),colour="#62D4D2",linewidth=.18) +
    geom_sf(data=s,aes(fill=station_id),shape=21,colour="white",size=3.1,stroke=.75) +
    geom_text_repel(data=s_xy,aes(X,Y,label=station_id),colour="white",fontface="bold",size=2.05,
                    seed=24,box.padding=.2,point.padding=.3,min.segment.length=0,
                    segment.colour="white",segment.size=.25,max.overlaps=Inf) +
    scale_fill_manual(values=station_colours,guide="none") +
    coord_sf(xlim=bb[c("xmin","xmax")],ylim=bb[c("ymin","ymax")],expand=FALSE,datum=NA) +
    annotate("label", x=bb[["xmin"]]+120, y=bb[["ymax"]]-120, label=panel_letter,
             hjust=0, vjust=1, colour="black", fontface="bold", size=2.7,
             fill=scales::alpha("white",.86), label.size=0,
             label.padding=unit(1.1,"mm")) +
    annotate("label", x=bb[["xmax"]]-120, y=bb[["ymax"]]-120, label=cluster_label,
             hjust=1, vjust=1, colour="black", fontface="bold", size=2.35,
             fill=scales::alpha("white",.86), label.size=0,
             label.padding=unit(.9,"mm")) +
    theme_void(base_size=7.2) +
    theme(panel.border=element_rect(colour="#333333",fill=NA,linewidth=.4),
          plot.margin=margin(0,0,0,0))
}

2 Introduction

Seagrasses are marine angiosperms that form highly productive meadows across shallow coastal waters (Orth et al., 2006). By creating three-dimensional habitat, attenuating waves, stabilising sediments and recycling nutrients, these foundation species support diverse assemblages of invertebrates, fishes, birds and marine megafauna (Waycott et al., 2009). Their ecological functions translate into services that include nursery habitat and fisheries support, shoreline protection, water-quality regulation and long-term organic-carbon storage (Barbier et al., 2011; Fourqurean et al., 2012; Nordlund et al., 2016; Unsworth et al., 2019). Yet seagrass meadows are exposed to multiple, often interacting pressures, including coastal development, eutrophication, altered sediment regimes, destructive fishing and anchoring, and climate-driven warming and extreme events (Orth et al., 2006; Waycott et al., 2009). A synthesis of 547 sites estimated a net loss of 19.1% of the surveyed meadow area since 1884, while also revealing strong regional heterogeneity and recent recovery at some locations (Dunic et al., 2021). This variability makes sustained, spatially explicit observation essential: global trends cannot substitute for local evidence, and recovery cannot be distinguished from continued degradation without repeatable measurements (Duffy et al., 2019). The Parc National du Banc d’Arguin (PNBA), Mauritania, is an exceptional case. Located at the interface between the Sahara and the eastern boundary upwelling system of the North Atlantic, this approximately 12,000-km² protected seascape contains one of the largest remaining seagrass systems in Africa (Pottier et al., 2021; Wolff et al., 1993). A recent park-wide habitat map estimated about 772 km² of seagrass, comprising approximately 353 km² of intertidal meadow dominated by Zostera noltei and 419 km² of predominantly subtidal meadow dominated by Cymodocea nodosa (Pottier et al., 2021). These meadows sustain the benthic food web, provide habitat and nursery grounds for fishes and invertebrates, and underpin a site used by more than two million migratory waterbirds along the East Atlantic Flyway (El-Hacen et al., 2020; van der Heide et al., 2012). Their extent does not imply ecological stability: between spatially comprehensive surveys conducted in 1986 and 2014, seagrass cover increased approximately twofold while macrozoobenthic community composition and estimated secondary production changed markedly (El-Hacen et al., 2020). The Banc d’Arguin meadow is therefore both a globally important conservation asset and a dynamic ecosystem whose trajectories require reliable observation (de Fouw et al., 2016; El-Hacen et al., 2020).

The environmental setting of Banc d’Arguin is governed by the juxtaposition of a hyper-arid desert margin, broad shallow tidal flats and the wind-driven Mauritanian upwelling system (Ould Dedah, 1993; Wolff et al., 1993). Freshwater input is negligible, precipitation is generally below 100 mm yr⁻¹, and persistent northerly winds help control waves, currents, sediment redistribution and exchanges between the bank and adjacent shelf (Ould Dedah, 1993; Wolff et al., 1993). Wind forcing is strongly seasonal: mean speeds reported for the intertidal system rise to about 9 m s⁻¹ in May–June and decline to about 6 m s⁻¹ in November–December (Ould Dedah, 1993). This annual cycle is embedded within the latitudinal migration of the Intertropical Convergence Zone, which reorganises both the origin and vertical pathway of mineral-dust transport over northwestern Africa (Knippertz and Todd, 2012). During winter, dust reaching coastal Mauritania is transported mainly in low-level north-easterly trade winds from Western Saharan and regional sources; during summer, transport increasingly occurs within the elevated Saharan Air Layer and draws on sources in Mauritania, Mali and farther east (Friese et al., 2017; Senghor et al., 2017). Superimposed on this seasonality are discrete dust outbreaks, including events that deliver anomalously coarse particles offshore (Friese et al., 2016). This aerosol regime creates a particular obstacle to optical monitoring of Banc d’Arguin, where low-tide habitat mapping depends on preserving the spectral consistency of exposed intertidal surfaces across acquisition dates (Song et al., 2001). Each usable acquisition captures a highly heterogeneous coastal mosaic comprising dense and sparse Zostera noltei, wet and dry sediment, tidal pools and adjacent bright desert surfaces (Pottier et al., 2021). Above this mosaic, mineral-dust abundance, particle-size distribution, source region and absorption properties vary seasonally and among individual outbreaks (Friese et al., 2016; Kaufman et al., 2002; Knippertz and Todd, 2012). These changes can alter the apparent brightness and spectral shape of otherwise unchanged intertidal targets, producing both spatial gradients within a scene and radiometric offsets among acquisition dates (Doxani et al., 2023; Song et al., 2001). The difficulty is compounded by the way Sen2Cor, the atmospheric correction algorithm routinely applied on Sentinel-2 images by the European Space Agency, estimates aerosol optical thickness (Main-Knorn et al., 2017). The processor preferentially retrieves aerosol information over reference targets with known spectral behaviour, particularly dense dark vegetation, and uses fallback information or default values where suitable reference areas are unavailable (European Space Agency, 2021; Main-Knorn et al., 2017). In the hyper-arid Banc d’Arguin landscape, dominated by bright desert surfaces and an intertidal zone that alternates among vegetation, moist sediment, bare sediment and shallow pools, the availability and representativeness of such reference targets cannot be assumed (Pottier et al., 2021; Wolff et al., 1993). An aerosol model adequate for one acquisition may consequently leave a different residual bias in another (Doxani et al., 2023). Thick, absorbing Saharan dust has already been shown to defeat conventional atmospheric correction off western Africa (Moulin et al., 2001), while comparative evaluations demonstrate that processor performance is strongly environment-dependent (Bui et al., 2022; Doxani et al., 2023; Warren et al., 2019). For this study, the central requirement is therefore not merely to retrieve plausible reflectance for a single date, but to preserve the field-measured spectral relationships among exposed intertidal targets across acquisitions made under contrasting dust conditions (Doxani et al., 2023; Song et al., 2001).

Mapping exposed intertidal Zostera noltei requires stable spectral relationships among seagrass canopies, wet and dry sediment, macroalgae and residual tidal pools. Discrimination is particularly challenging where seagrasses and co-occurring green macroalgae, which share major photosynthetic pigments, produce partly overlapping multispectral signatures (Davies et al., 2023). Nevertheless, spectral-library experiments have shown that machine-learning classifiers can distinguish soft-bottom intertidal vegetation at Sentinel-2 spectral resolution, with little improvement obtained by increasing spectral resolution beyond approximately ten bands (Davies et al., 2023). Building on this foundation, Davies et al. developed the Intertidal Classification of Europe: Categorising Reflectance of Emerged Areas of Marine vegetation with Sentinel-2 (ICE CREAMS v1.0), a multiclass neural network trained with extensive drone-derived labels to identify seagrass and other exposed intertidal habitats (Davies et al., 2024a). Independent validation across approximately 12,000 Sentinel-2 pixels yielded an overall accuracy of 0.82 for seagrass presence or absence, and subsequent applications resolved both seasonal phenology across 12 North-East Atlantic meadows and contrasting interannual trajectories at six European sites (Davies et al., 2024a, 2024b). These results establish ICE CREAMS as a scalable framework for intertidal seagrass monitoring, but transferring the classifier among dates and regions presupposes that the spectral relationships encoded by the model are preserved in the input imagery (Davies et al., 2024a, 2024b; Song et al., 2001). Under the variable aerosol loads of Banc d’Arguin, incomplete correction can alter both the magnitude and shape of target spectra (Doxani et al., 2023; Song et al., 2001). A classifier trained on one radiometric condition may then encounter another: unchanged seagrass can migrate in feature space, while atmospheric variability can be misinterpreted as habitat gain, loss or change in cover (Song et al., 2001). This problem is particularly consequential for time-series analysis, for which consistency among dates is at least as important as the accuracy of any single scene (Lyons et al., 2013; Roelfsema et al., 2014). Sentinel-2 has already demonstrated the feasibility of mapping the PNBA’s coastal habitats, with a single-date supervised classification attaining more than 80% overall accuracy (Pottier et al., 2021). However, a static inventory cannot establish the timing, persistence or spatial pattern of ecological change (Davies et al., 2024b; Lyons et al., 2013). Conversely, the pronounced differences observed between the 1986 and 2014 benthic surveys demonstrate the value of long-term comparison but do not resolve the intervening trajectory (El-Hacen et al., 2020). A radiometrically consistent satellite record could bridge these scales by tracking expansion, contraction and fragmentation at frequencies unattainable through field surveys alone (Davies et al., 2024a, 2024b). Without such a record, managers cannot confidently separate ecological change from retrieval artefacts, establish contemporary baselines, identify persistent hotspots of loss, evaluate conservation measures or target field surveillance (Duffy et al., 2019; Unsworth et al., 2019). More broadly, incomplete information on meadow status and condition remains a major barrier to seagrass conservation, and coordinated observing frameworks explicitly call for the integration of field measurements with repeated satellite mapping (Duffy et al., 2019; Unsworth et al., 2019). Resolving atmospheric-correction uncertainty is thus a prerequisite not only for classification accuracy, but also for defensible conservation assessments and policy decisions in the PNBA (Doxani et al., 2023; Song et al., 2001).

This study addresses that bottleneck by establishing a locally validated workflow for mapping the intertidal seagrass meadows of Banc d’Arguin through time. First, we use field hyperspectral measurements acquired over representative seagrass and non-seagrass targets to evaluate Sentinel-2 reflectance produced by Sen2Cor and two alternative atmospheric-correction approaches. The field spectra are matched to the Sentinel-2 spectral response, enabling us to assess both spectral fidelity and temporal consistency and to identify the correction best suited to this dust-influenced coastal environment. Second, we adapt the ICE CREAMS classification framework to the selected reflectance product, so that the habitat model operates on radiometrically coherent inputs rather than on correction-dependent spectral artefacts. Third, we apply the resulting workflow to the Sentinel-2 archive to reconstruct a time series of seagrass distribution and quantify the spatial and temporal dynamics of the intertidal meadow. The following sections describe the field campaign, satellite processing and atmospheric-correction comparison, the adapted classification workflow, and the procedures used to validate maps and derive change metrics.

3 Materials and Methods

3.1 Study area and field reflectance measurements

The study was conducted in the Parc National du Banc d’Arguin (PNBA), north-western Mauritania, an extensive system of islands, shallow channels and intertidal flats at the interface between the Sahara and the eastern Atlantic (Pottier et al., 2021; Wolff et al., 1993). Fieldwork took place on 20–30 March 2026. Surface reflectance was measured with a portable ASD field spectroradiometer over 400–900 nm. On 27 March 2026, coincident with a Sentinel-2B overpass, eight stations (AC01–AC08) were sampled at two sites about 22 km apart (Figure 1): two on dry coastal sand (AC01–AC02), two on microphytobenthos (MPB) mud (AC03–AC04), one on algal mud (AC05), one on white sand (AC06) and two on Zostera noltei seagrass (AC07–AC08). At each station, spectra were recorded within an area of about one 10 m Sentinel-2 pixel (five sets of 20 spectra), and the average spectrum of the station was used as the field reference.

Each station spectrum was convolved to the bands of the Sentinel-2B Multispectral Instrument (MSI) that lie within the ASD range (B1–B8A; nine bands) using the official spectral response functions \(S_b(\lambda)\) (ESA technical note COPE-GSEG-EOPG-TN-15-0007, version 4.0):

\[ R_b = \frac{\sum_{\lambda} R(\lambda)\,S_b(\lambda)}{\sum_{\lambda} S_b(\lambda)}, \]

where \(R(\lambda)\) is the field reflectance at 1 nm and the sums run over the wavelengths covered by the ASD (Figure 2). Weighting the response functions by the extraterrestrial solar spectrum (Wehrli, 1985) changed the root-mean-square deviation (RMSD) of either processor by less than 0.0001 and is not used further.

3.2 Satellite reflectance products

Two surface-reflectance products were derived for the Sentinel-2B acquisition of 27 March 2026 (sensed at 11:45:53 UTC; tile 28QCH; processing baseline 05.12; cloud cover 0.015% of the tile). The first was the operational Level-2A product generated with Sen2Cor (European Space Agency, 2021; Main-Knorn et al., 2017), in which digital numbers were converted to bottom-of-atmosphere reflectance as \((\mathrm{DN} - 1000)/10{,}000\). Bands were used on their native grids (10 m: B2, B3, B4, B8; 20 m: B5, B6, B7, B8A; 60 m: B1). The second was obtained by processing the corresponding Level-1C product with ACOLITE using dark spectrum fitting (Vanhellemont, 2019; Vanhellemont and Ruddick, 2018). Aerosol optical thickness was estimated in tiles of 600 × 600 pixels with minimum-DRMSD selection of the aerosol model and a three-pixel smoothing kernel. Surface reflectance (rhos) was produced at 10 m resolution, without adjacency or glint correction. ACOLITE bands were assigned to MSI bands according to the equivalent Sentinel-2B central wavelengths (maximum difference 0.34 nm).

3.3 Match-up of field and satellite reflectance

Station coordinates were projected to the coordinate system of each raster, and satellite reflectance was extracted from the pixel containing each station. The match-up comprised 72 paired values per processor (eight stations × nine bands). With \(F_i\) the convolved field reflectance, \(S_i\) the satellite reflectance and \(d_i = S_i - F_i\) the residual, agreement was quantified by the RMSD and the mean bias,

\[ \mathrm{RMSD}=\sqrt{\tfrac{1}{n}\sum_i d_i^2},\qquad \mathrm{bias}=\tfrac{1}{n}\sum_i d_i , \]

computed over all paired values, by band, by station and by surface class. Spectral-shape agreement was measured for each station by the spectral angle between the nine-band satellite and field vectors. The difference between processors was expressed as \(\Delta\mathrm{RMSD} = \mathrm{RMSD}_{\mathrm{Sen2Cor}} - \mathrm{RMSD}_{\mathrm{ACOLITE}}\), so that positive values indicate a lower RMSD for ACOLITE.

Because the nine bands of a station are not independent, uncertainty was estimated by a station-cluster percentile bootstrap (Field and Welsh, 2007): stations were resampled with replacement, all bands of a sampled station were kept together, and RMSD, bias and \(\Delta\)RMSD were computed on the same draws for both processors (10,000 draws). The 2.5th and 97.5th percentiles are reported as 95% intervals. The influence of individual stations was assessed by recomputing \(\Delta\)RMSD after omitting each station in turn.

Three analytical choices were varied to test the robustness of the comparison: (i) exclusion of AC03, whose spectrum differs markedly between field and satellite products (see Results), giving a seven-station set (63 paired values per processor); (ii) extraction of both products on a common 10 m grid, using nearest-neighbour and bilinear sampling; and (iii) replacement of the single pixel by the mean of the 3 × 3 pixels (30 m) centred on each station.

3.4 Pixel-wise comparison of Sen2Cor and ACOLITE reflectance

To describe how the two processors differ across the landscape, independently of the field stations, the Sen2Cor and ACOLITE reflectance of the 27 March 2026 scene were compared pixel by pixel for the eleven bands common to both products (B1–B8, B8A, B11 and B12). B9 was not available in the ACOLITE export and B10 is not a surface-reflectance band. Comparisons were made on the 10 m ACOLITE grid of the study area (3,763 × 1,974 pixels, WGS 84 / UTM zone 28N). Sen2Cor reflectance was resampled to this grid by bilinear interpolation from its native 10, 20 or 60 m resolution, and the Sen2Cor scene classification layer (SCL) by nearest-neighbour assignment. Digital numbers were converted to reflectance as described above. The difference was defined as \(\Delta\rho = \rho_\mathrm{ACOLITE} - \rho_\mathrm{Sen2Cor}\), so that positive values indicate higher ACOLITE reflectance.

Pixels were retained when the Sen2Cor digital number was positive, the ACOLITE value was valid, and the SCL class was not no-data, saturated or defective, cloud shadow, cloud, cirrus or snow/ice. Water, vegetation, bare-surface and other clear-sky classes were retained and no water-only mask was applied. Negative reflectance values were kept, because removing them would hide a feature of the products under comparison. For each band, the distribution of \(\Delta\rho\) was summarised by its mean, median, standard deviation and percentiles, and by the proportion of pixels with \(|\Delta\rho|\) above 0.01, 0.02, 0.05 and 0.10. To examine the spatial organisation of the differences, the most divergent 0.1% of pixels in each band (largest \(|\Delta\rho|\)) were assigned to mutually exclusive strata: the intertidal zone (a polygon layer delineating the intertidal flats), SCL water, SCL vegetation and SCL bare surfaces outside that polygon, and other pixels. The share of each stratum in the scene and the exceedance rate within each stratum were also computed, so that the composition of the extreme pixels could be read against the area covered by each stratum. Dependence of \(\Delta\rho\) on surface brightness was examined by binning pixels according to Sen2Cor reflectance. As a check on resampling and edge effects, the analysis was repeated for the four native 10 m bands (B2, B3, B4 and B8) separately for SCL water and non-water pixels, after eroding the valid-pixel mask by one 10 m pixel.

Six 3 × 3-pixel windows (30 × 30 m) were selected to illustrate paired spectral signatures. They were chosen by fixed rules applied to the image: the window of minimum Sen2Cor near-infrared (B8) reflectance among SCL water pixels (T1); the window of maximum Sen2Cor red (B4) reflectance among SCL water pixels (T2); the windows of minimum B8 and of maximum B4 reflectance within the intertidal polygon (T3 and T4); the window of maximum Sen2Cor NDVI among SCL vegetation pixels (T5); and the window of maximum B4 reflectance among SCL bare-surface pixels (T6). The window means of ACOLITE and Sen2Cor reflectance were compared band by band. The targets are image-derived strata and carry no field-verified habitat labels. No reference measurement is available for this comparison, which therefore describes differences between the two processors and does not assess the accuracy of either. Calculations were performed in Python 3.12 with NumPy 1.26 and GDAL 3.8.

4 Results

4.1 Match-ups of atmospheric corrections

Both products reproduced the large reflectance contrast among surfaces (Figure 3). Over the 72 paired values, ACOLITE had an RMSD of 0.0500 (95% interval 0.0227–0.0698) and a mean bias of +0.0032 (−0.0258 to +0.0312), whereas Sen2Cor had an RMSD of 0.0659 (0.0402–0.0851) and a mean bias of +0.0309 (−0.0037 to +0.0635). The paired difference was \(\Delta\)RMSD = +0.0159 (+0.0017 to +0.0357); the intervals of the two processors overlapped broadly.

The difference between processors was expressed mainly as a bias. Sen2Cor reflectance was higher than the field reflectance in all nine bands (mean bias +0.013 in B8A to +0.054 in B5; band RMSD 0.027–0.085), whereas ACOLITE biases were smaller and of mixed sign (−0.006 to +0.021; band RMSD 0.022–0.072) (Figure 4). ACOLITE had the lower RMSD at six of the eight stations (AC01, AC02, AC04, AC05, AC06 and AC08); Sen2Cor had the lower RMSD at AC07 (0.0126 versus 0.0219) and, marginally, at AC03 (0.0928 versus 0.0968). The contrast was concentrated at the brightest surfaces. Over white sand (AC06), Sen2Cor had an RMSD of 0.0950 and a bias of +0.0900, against 0.0229 and −0.0044 for ACOLITE; omitting AC06 reduced \(\Delta\)RMSD from +0.0159 to +0.0079, whereas omitting any other station gave +0.0152 to +0.0222. Both processors were positively biased over dry sand (Sen2Cor +0.072; ACOLITE +0.051). Over the two seagrass stations, RMSD was 0.0331 for ACOLITE and 0.0403 for Sen2Cor. At these stations the mean ACOLITE residual was positive in B1–B5 (+0.011 to +0.046) and negative in B6–B8A (−0.015 to −0.024), whereas the mean Sen2Cor residual was positive in B1–B5 (+0.014 to +0.054) and within ±0.009 in B6–B8A. Values for each surface class are given in Table 1.

Spectral shape was preserved at most stations. The median spectral angle between the satellite and field spectra was 2.60° for ACOLITE and 2.83° for Sen2Cor (2.28° and 2.78° without AC03). Two stations departed from this pattern for both processors: AC03 (17.0° for ACOLITE; 20.5° for Sen2Cor) and AC08 (12.6°; 13.7°). At AC03, residuals in B1–B5 were between −0.050 and +0.002, but residuals in B6–B8A were between −0.12 and −0.16 for both processors, the two products returning similar reflectance (about 0.09–0.14) against a field reflectance of about 0.25. At AC08, satellite reflectance in B3 was 0.148–0.169 against a field value of 0.086, while B8A agreed closely (0.240–0.256 against 0.248).

4.2 Sensitivity of the match-up to analytical choices

Excluding AC03 lowered the RMSD of both processors (ACOLITE 0.0389, 0.0191–0.0578; Sen2Cor 0.0611, 0.0314–0.0826) and increased their mean biases (+0.0152 and +0.0449, respectively), while \(\Delta\)RMSD remained positive (+0.0222; +0.0075 to +0.0432). Extraction on a common 10 m grid left the result for ACOLITE unchanged (RMSD 0.0500) and gave an RMSD for Sen2Cor of 0.0659 with nearest-neighbour and 0.0680 with bilinear sampling. Using the mean of 3 × 3 pixels gave an RMSD of 0.0525 for ACOLITE and 0.0672 (nearest neighbour) and 0.0645 (bilinear) for Sen2Cor. The sign of \(\Delta\)RMSD was positive in all variants.

4.3 Pixel-wise comparison of Sen2Cor and ACOLITE reflectance

Approximately 7.31 million pixels per band were valid in both products. The scene-wide mean of \(\Delta\rho\) was slightly negative in every band, from −0.0009 (B8A) to −0.0089 (B05), and its standard deviation ranged from 0.0072 (B1) to 0.0167 (B12). Absolute differences exceeded 0.01 in 14.8% (B8A) to 33.0% (B2) of pixels, exceeded 0.05 in 0.03% (B1) to 7.5% (B12) of pixels, and exceeded 0.10 in no more than 0.015% of pixels in any band. The small scene-wide means conceal marked spatial structure (Figure 5). In all bands the differences were spatially coherent rather than random, and were organised along coastlines, tidal channels and the margins of the intertidal flats, where narrow bands of positive \(\Delta\rho\) lay beside areas of negative \(\Delta\rho\) in the visible and near-infrared bands. Negative differences of largest magnitude occurred over extensive areas in the south and east of the scene, most clearly in B5, B11 and B12.

The composition of the largest differences reflected this pattern. In B1–B7, 69.2–91.9% of the most divergent 0.1% of pixels were SCL bare surfaces outside the intertidal polygon and 5.7–27.4% lay within it, although bare surfaces covered 16.2% of valid pixels, the intertidal polygon 33.1% and SCL water outside the polygon 50.7%. The within-stratum exceedance rate was accordingly 0.43–0.57% for bare surfaces and 0.005–0.083% for the water and intertidal strata. In B12, the mean \(\Delta\rho\) was −0.043 over bare surfaces and −0.002 over both the intertidal polygon and SCL water. The sign of the difference depended on surface brightness. In B4, the mean \(\Delta\rho\) was +0.032 where Sen2Cor reflectance was negative, +0.012 where it was between 0 and 0.01, and −0.012 where it exceeded 0.4. The corresponding values in B8 were +0.019, +0.005 and −0.024. Sen2Cor produced negative reflectance in 0% (B11) to 0.86% (B6) of pixels, equivalent to 62,529 pixels in B6, whereas ACOLITE produced at most 72 negative pixels in any band. When the valid-pixel mask was eroded by one pixel, the mean \(\Delta\rho\) of the four native 10 m bands remained negative for both SCL water (−0.0019 in B4 to −0.0069 in B2) and non-water pixels (−0.0060 in B3 and B4 to −0.0173 in B8).

The paired spectra of the six target windows are shown in Figure 6. Over the dark targets T1 (dark SCL water) and T3 (dark intertidal), ACOLITE reflectance exceeded Sen2Cor reflectance by 0.015–0.033 from B1 to B8A, with smaller differences in B11 (0.005–0.006) and B12 (0.002–0.003). Sen2Cor reflectance was negative at T1 in B4–B8A (down to −0.013 in B6) and at T3 in B6–B8A (down to −0.011 in B6), whereas ACOLITE reflectance was positive in all bands at both targets. Over the bright targets T2, T4 and T6, ACOLITE reflectance was lower than Sen2Cor reflectance in every band. The differences were largest in the red-edge and visible bands (T6, −0.154 in B5; T2, −0.120 in B2; T4, −0.086 in B5) and smallest in B1 for T4 (−0.008) and in B12 for T2 and T6 (−0.028 and −0.060). The shapes of the two spectra were similar at these targets, with the same increase from blue to near-infrared and the same decline in the shortwave infrared. At the vegetation target T5, both products showed the red-edge increase and near-infrared plateau, with ACOLITE reflectance higher by 0.019–0.028 in B1–B4 and by 0.001–0.010 in B5–B12 (0.002 in B8).
